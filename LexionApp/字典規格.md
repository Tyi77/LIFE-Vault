```SQL
CREATE TABLE lexicon (
    -- 1. SN: Strong Number (主鍵，用於排序)
    -- 格式範例： "02316" (補零以確保排序正確：00001, 00002... 02316)
    sn TEXT PRIMARY KEY, 
    
    -- 2. 顯示用希臘文 (含變音)
    greek_text TEXT,              -- θεός
    
    -- 3. 搜尋用欄位
    greek_normalized TEXT,        -- θεος (去變音，供希臘文鍵盤輸入)
    transliteration TEXT,         -- theos (新增：供英文鍵盤輸入發音)
    
    -- 4. 內容欄位 (從 simplified content 拆解)
    lemma TEXT,                   -- 原型
    pos TEXT,                     -- N-N M
    gloss TEXT,                   -- 神, 上帝 (中文簡義)
    
    -- 5. 備註欄位 (存希臘文)
    remarks TEXT,                 -- τίθημι (直接存希臘字)

    -- 6. 完整內容 (HTML)
    full_content TEXT
);
```
